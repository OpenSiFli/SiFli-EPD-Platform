cmake_minimum_required(VERSION 3.22)
project(unified_lv_context C)
set(CMAKE_C_STANDARD 11)
option(CONTEXT_ENABLED "Enable LVGL execution context" ON)
option(CONTEXT_ASAN "Enable address sanitizer" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVGL "${ROOT}/SiFli-SDK/external/lvgl_v9")
file(GLOB_RECURSE LV_SOURCES CONFIGURE_DEPENDS "${LVGL}/src/*.c")
add_library(lvgl_host STATIC ${LV_SOURCES})
target_include_directories(lvgl_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${LVGL}")
target_compile_definitions(lvgl_host PUBLIC LV_CONF_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS)
if(CONTEXT_ENABLED)
  target_compile_definitions(lvgl_host PUBLIC LV_USE_EXECUTION_CONTEXT=1)
endif()
add_executable(context_test host.c)
target_link_libraries(context_test PRIVATE lvgl_host)
set(INPUT_SOURCE "${ROOT}/src/ui/platform/epd_input.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${INPUT_SOURCE}")
file(READ "${INPUT_SOURCE}" INPUT_CODE)
string(REGEX REPLACE "#include[^\n]*\n" "" INPUT_CODE "${INPUT_CODE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/input_impl.inc" "${INPUT_CODE}")
add_executable(input_context_test input_host.c)
target_include_directories(input_context_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
  "${CMAKE_CURRENT_SOURCE_DIR}/input_stubs" "${ROOT}/src" "${ROOT}/src/platform")
target_link_libraries(input_context_test PRIVATE lvgl_host)
if(MSVC)
  target_compile_options(lvgl_host PRIVATE /utf-8 /MP /wd4244 /wd4267)
  target_compile_options(context_test PRIVATE /utf-8 /W4 /wd4244 /wd4267)
  target_compile_options(input_context_test PRIVATE /utf-8 /W4 /wd4244 /wd4267)
  if(CONTEXT_ASAN)
    target_compile_options(lvgl_host PRIVATE /fsanitize=address)
    target_compile_options(context_test PRIVATE /fsanitize=address)
    target_link_options(context_test PRIVATE /INCREMENTAL:NO)
    target_compile_options(input_context_test PRIVATE /fsanitize=address)
    target_link_options(input_context_test PRIVATE /INCREMENTAL:NO)
    get_filename_component(ASAN_RUNTIME_DIR "${CMAKE_C_COMPILER}" DIRECTORY)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
      set(ASAN_RUNTIME_ARCH x86_64)
    else()
      set(ASAN_RUNTIME_ARCH i386)
    endif()
    set(ASAN_RUNTIME_DLL "${ASAN_RUNTIME_DIR}/clang_rt.asan_dynamic-${ASAN_RUNTIME_ARCH}.dll")
    if(NOT EXISTS "${ASAN_RUNTIME_DLL}")
      message(FATAL_ERROR "AddressSanitizer runtime missing from the selected compiler: ${ASAN_RUNTIME_DLL}")
    endif()
    set(ASAN_RUNTIME_DLLS "${ASAN_RUNTIME_DLL}")
    set(ASAN_DEBUG_DLL "${ASAN_RUNTIME_DIR}/clang_rt.asan_dbg_dynamic-${ASAN_RUNTIME_ARCH}.dll")
    if(EXISTS "${ASAN_DEBUG_DLL}")
      list(APPEND ASAN_RUNTIME_DLLS "${ASAN_DEBUG_DLL}")
    endif()
    foreach(TEST_TARGET context_test input_context_test)
      add_custom_command(TARGET ${TEST_TARGET} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${ASAN_RUNTIME_DLLS}
          "$<TARGET_FILE_DIR:${TEST_TARGET}>"
        VERBATIM)
    endforeach()
  endif()
else()
  target_link_libraries(context_test PRIVATE m)
  target_link_libraries(input_context_test PRIVATE m)
  if(CONTEXT_ASAN)
    target_compile_options(lvgl_host PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_compile_options(context_test PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(context_test PRIVATE -fsanitize=address)
    target_compile_options(input_context_test PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(input_context_test PRIVATE -fsanitize=address)
  endif()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME execution_context COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/run_host_test.py" "$<TARGET_FILE:context_test>")
add_test(NAME input_context COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/run_host_test.py" "$<TARGET_FILE:input_context_test>")
set_tests_properties(execution_context input_context PROPERTIES TIMEOUT 60)
if(MSVC AND CONTEXT_ASAN)
  set_tests_properties(execution_context input_context PROPERTIES
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${ASAN_RUNTIME_DIR}")
endif()
add_test(NAME template_config COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/template_config_test.py")
add_test(NAME app_profile COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/app_profile_test.py")
