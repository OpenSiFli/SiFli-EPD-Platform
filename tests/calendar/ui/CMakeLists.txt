cmake_minimum_required(VERSION 3.20)
project(calendar_ui C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "../../.." ABSOLUTE)
set(LVGL "${REPO}/SiFli-SDK/external/lvgl_v9")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${GENERATED}/ui.inc" "${GENERATED}/runtime.h" "${GENERATED}/picker.inc" "${GENERATED}/time_settings.inc"
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${REPO}" "${GENERATED}"
 DEPENDS "${REPO}/src/platform/epd_date_picker.c" "${REPO}/src/ui/apps/app_time_settings.c" generate.py "${REPO}/modules/calendr/calendar_ui.c" "${REPO}/modules/calendr/calendar_runtime.h")
file(GLOB_RECURSE LV_SOURCES "${LVGL}/src/*.c")
list(FILTER LV_SOURCES EXCLUDE REGEX "/(draw/sifli|libs/rlottie|libs/ffmpeg|libs/gstreamer)/")
add_library(lvgl_host STATIC ${LV_SOURCES})
target_include_directories(lvgl_host PUBLIC "${LVGL}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl_host PUBLIC LV_CONF_INCLUDE_SIMPLE LV_KCONFIG_IGNORE _CRT_SECURE_NO_WARNINGS)
add_executable(calendar_ui host.c "${GENERATED}/ui.inc" "${GENERATED}/runtime.h" "${GENERATED}/picker.inc" "${GENERATED}/time_settings.inc" "${REPO}/modules/calendr/calendar_model.c" "${REPO}/src/platform/epd_calendar.c")
target_include_directories(calendar_ui PRIVATE "${GENERATED}" "${REPO}/modules/calendr" "${REPO}/src/platform" "${REPO}/src/ui")
target_compile_definitions(calendar_ui PRIVATE CAL_HOST EPD_CALENDAR_HOST)
target_link_libraries(calendar_ui PRIVATE lvgl_host)
if(MSVC)
 target_compile_options(lvgl_host PRIVATE /utf-8 /MP /wd4244 /wd4267)
 target_compile_options(calendar_ui PRIVATE /utf-8 /W3)
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/screens")
enable_testing()
add_test(NAME calendar_native_ui COMMAND calendar_ui "${REPO}/font/MiSans-Normal.ttf" "${CMAKE_CURRENT_BINARY_DIR}/screens")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/screens-partial")
add_test(NAME calendar_partial_rgb565 COMMAND calendar_ui "${REPO}/font/MiSans-Normal.ttf" "${CMAKE_CURRENT_BINARY_DIR}/screens-partial" partial)
add_test(NAME calendar_draft_failures COMMAND calendar_ui "${REPO}/font/MiSans-Normal.ttf" "${CMAKE_CURRENT_BINARY_DIR}/screens" draft-failures)
