cmake_minimum_required(VERSION 3.20)
project(calendar_tests C)
set(CMAKE_C_STANDARD 11)
add_executable(calendar_tests calendar_test.c ../../modules/calendr/calendar_store.c ../../modules/calendr/calendar_model.c ../../src/platform/epd_calendar.c)
