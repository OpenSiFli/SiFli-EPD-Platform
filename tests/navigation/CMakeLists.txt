cmake_minimum_required(VERSION 3.20)
project(navigation_tests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "../.." ABSOLUTE)
enable_testing()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../notes/ui" "notes")
add_executable(navigation_tests gesture_test.c
    "${REPO}/src/ui/ui_nav_gesture.c" "${REPO}/src/ui/ui_shell_model.c")
target_include_directories(navigation_tests PRIVATE "${REPO}/src/ui")
target_link_libraries(navigation_tests PRIVATE lvgl_host)
if(MSVC)
    target_compile_options(navigation_tests PRIVATE /utf-8 /UNDEBUG)
endif()
add_test(NAME navigation_gestures COMMAND navigation_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${GENERATED}/shell_actions.inc" "${GENERATED}/shell_edges.inc" "${GENERATED}/shell_lifecycle.inc"
    "${GENERATED}/render_profile_actual.inc" "${GENERATED}/ui_perf_actual.inc"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_inputs.py" "${REPO}" "${GENERATED}"
    DEPENDS generate_inputs.py "${REPO}/src/ui/ui_shell.c" "${REPO}/src/ui/platform/epd_app.c" "${REPO}/src/main.c")
target_sources(navigation_tests PRIVATE "${GENERATED}/shell_edges.inc")
target_include_directories(navigation_tests PRIVATE "${GENERATED}")
add_executable(navigation_routing routing_test.c "${GENERATED}/shell_actions.inc")
target_include_directories(navigation_routing PRIVATE "${GENERATED}" "${REPO}/src/ui")
target_link_libraries(navigation_routing PRIVATE lvgl_host)
if(MSVC)
    target_compile_options(navigation_routing PRIVATE /utf-8 /UNDEBUG /D_CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME navigation_routing COMMAND navigation_routing)

add_executable(navigation_lifecycle lifecycle_test.c "${GENERATED}/shell_lifecycle.inc")
target_include_directories(navigation_lifecycle PRIVATE "${GENERATED}")
target_link_libraries(navigation_lifecycle PRIVATE lvgl_host)
if(MSVC)
    target_compile_options(navigation_lifecycle PRIVATE /utf-8 /UNDEBUG)
endif()
add_test(NAME navigation_lifecycle COMMAND navigation_lifecycle)

add_executable(render_profile_tests render_profile_test.c "${GENERATED}/render_profile_actual.inc" "${GENERATED}/ui_perf_actual.inc")
target_include_directories(render_profile_tests PRIVATE "${GENERATED}" "${REPO}/src/ui/platform")
target_link_libraries(render_profile_tests PRIVATE lvgl_host)
if(MSVC)
    target_compile_options(render_profile_tests PRIVATE /utf-8 /W4 /WX /UNDEBUG)
endif()
add_test(NAME navigation_render_profile COMMAND render_profile_tests)

add_custom_command(OUTPUT "${GENERATED}/control_center_actual.inc"
    COMMAND "${Python3_EXECUTABLE}" "${REPO}/tests/control_center/generate_inputs.py" "${REPO}" "${GENERATED}"
    DEPENDS "${REPO}/tests/control_center/generate_inputs.py" "${REPO}/src/ui/ui_control_center.c")
add_executable(control_center_tests "${REPO}/tests/control_center/host_test.c" "${GENERATED}/control_center_actual.inc")
target_include_directories(control_center_tests PRIVATE "${GENERATED}" "${REPO}/src/ui")
target_link_libraries(control_center_tests PRIVATE lvgl_host)
if(MSVC)
    target_compile_options(control_center_tests PRIVATE /utf-8 /UNDEBUG)
endif()
add_test(NAME control_center_lifecycle COMMAND control_center_tests)
